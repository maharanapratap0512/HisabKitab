What is sutramCore?

sutramCore is a lightweight, schema-driven SQLite ORM. You define your tables once using defineTable and col builders — then get a full CRUD API with joins, pagination, and CTE reports automatically.

Everything is synchronous — no async/await needed. If you need MySQL, use sutramcore-mysql which has the same API but async.

Connect

Create one Sutram instance per database. Two options — pass your own db, or let sutramCore open the file.

Option A — pass existing dbjs
const { Sutram } = require('sutramcore');
const schema = require('./schema');
const reports = require('./reports');

const sewa = new Sutram({
  db: dbmodal.db,
  schema,
  reports,   // optional
});
Option B — sutramCore opens filejs
const sewa = new Sutram({
  dbPath:     './app.db',
  schema,
  reports,
  migrations: require('./migrations'),
  views:      [],
});
04

defineTable

database/schema.jsjs
const { defineTable, col } = require('sutramcore');

module.exports = {

  ...defineTable('category', {
    _id:    col.id(),
    name:   col.string().required(),
    active: col.boolean().default(1),
  }),

  ...defineTable('product', {
    _id:         col.id(),
    name:        col.string().required(),
    price:       col.number().default(0),
    category_id: col.ref('category._id', {   // hasOne
                     as: 'category', select: ['name'],
                 }),
    meta:        col.json(),
    active:      col.boolean().default(1),
  }, {
    reviews: {                                   // hasMany
      hasMany: true, table: 'review',
      on: 'product_id', target: '_id',
      as: 'reviews', select: ['_id', 'rating'],
    },
    tags: {                                      // manyToMany
      manyToMany: true, table: 'tag',
      junction: 'product_tag',
      on: 'product_id', target: 'tag_id',
      as: 'tags', select: ['_id', 'label'],
    },
  }),

};
04b

defineView

Declare a SQLite VIEW backed by a table's full join query — in one line. The view bakes in all hasOne, hasMany, and manyToMany joins. You can then reference the view with col.ref() in another table, enabling unlimited join depth with zero custom SQL.

database/schema.jsjs
const { defineTable, defineView, col } = require('sutramcore');

module.exports = {

  // ── Level 1: base table — item joins unit (hasOne)
  ...defineTable('item', {
    _id:     col.id(),
    name:    col.string().required(),
    unit_id: col.ref('unit._id', { as: 'unit', select: ['unit_short'] }),
  }),

  // ── Level 2: view — bakes item + unit into a SQLite VIEW
  ...defineView('v_item', 'item'),
  //              ↑ view name   ↑ source table (uses its full join query)

  // ── Level 3: outer table joins the VIEW — gets item + unit in one shot
  ...defineTable('batch_item', {
    _id:      col.id(),
    batch_id: col.number(),
    item_id:  col.ref('v_item._id', { as: 'item', select: ['name', 'unit'] }),
    //                   ↑ join the VIEW — 'unit' is the nested object
    qty:      col.number(),
  }),

};

Result

sewa.table('batch_item').getById(1)js
// {
//   _id: 1, batch_id: 3, qty: 5,
//   item: {                        ← from v_item
//     name: 'Rice',
//     unit: { unit_short: 'kg' }   ← nested — from item's own join
//   }
// }

Unlimited depth — chain views

Each defineView() can be the source for the next one. Declare source before view — order in schema.js matters.

chaining viewsjs
// depth 1 — item + unit baked in
...defineView('v_item', 'item'),

// depth 2 — batch_item joins v_item → gets item + unit
...defineView('v_batch_item', 'batch_item'),

// depth 3 — order_line joins v_batch_item → gets everything, 3 levels deep
...defineTable('order_line', {
  batch_item_id: col.ref('v_batch_item._id', {
    as: 'batch_item', select: ['qty', 'item'],
  }),
}),
✓Declare defineTable() for the source BEFORE defineView() that references it — insertion order in schema.js is the resolution order.
⚠Views are read-only. Calling insert() / update() on a view will throw a SQLite error. Always write to the real underlying table.
ParameterTypeDescription
viewNamestringName for the SQLite VIEW — e.g. 'v_item'
sourceTableNamestringTable whose full join query backs the view

Migrations

Array-of-arrays. Each version is one array. Never edit past versions — add new arrays at the end. Tracked via user_version pragma.

database/migrations.jsjs
module.exports = [
  // version 1
  [
    `CREATE TABLE IF NOT EXISTS category (
      _id  INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      active TINYINT DEFAULT 1
    )`,
    `CREATE TABLE IF NOT EXISTS product (
      _id         INTEGER PRIMARY KEY AUTOINCREMENT,
      name        TEXT NOT NULL,
      price       DECIMAL(10,2) DEFAULT 0,
      category_id INTEGER REFERENCES category(_id),
      active      TINYINT DEFAULT 1
    )`,
  ],
  // version 2 — add column
  [
    `ALTER TABLE product ADD COLUMN sku TEXT`,
  ],
];

insert

insert examplesjs
const product = sewa.table('product');

// full: true (default) — returns complete row with joins
const row = product.insert({
  name:        'Wireless Mouse',
  price:       799,
  category_id: 2,
});
// row.category = { name: 'Electronics' }  ← joined

// full: false — returns insertId only (faster, use in transactions)
const id = product.insert({ name: 'USB Hub' }, false);

getById / getOne / getAll

get examplesjs
const product = sewa.table('product');

// getById — full with joins
const row = product.getById(1);

// getOne — first match
const row = product.getOne({ name: 'USB Hub' });
const row = product.getOne('price < 500 AND active = 1');

// getAll — filter + options
const rows = product.getAll({ active: 1 });
const rows = product.getAll({ active: 1 }, {
  orderBy: 'price ASC',
  limit:   20,
  offset:  0,
  full:    false,   // skip joins
});

CTE Reports

Named SQL queries — any SELECT, CTE, aggregation — defined in reports.js and executed via procedure(). CTE only in sutramCore (SQLite). For stored procedures, use sutramcore-mysql.

database/reports.jsjs
module.exports = [
  {
    report_name: 'monthly_sales',
    label:       'Monthly Sales Summary',
    report_type: 'cte',
    cte_sql: `
      SELECT strftime('%Y-%m', date) as month,
             COUNT(*) as orders,
             SUM(amount) as revenue
      FROM orders
      WHERE date >= :from AND date <= :to
      GROUP BY month ORDER BY month DESC
    `,
    params: [
      { name: 'from', type: 'string', default: '2024-01-01' },
      { name: 'to',   type: 'string', default: '2024-12-31' },
    ],
  },

  {
    report_name: 'low_stock',
    label:       'Low Stock Products',
    report_type: 'cte',
    cte_sql: `SELECT * FROM product WHERE stock < :threshold AND active = 1`,
    params: [
      { name: 'threshold', type: 'number', default: 10 },
    ],
  },
];
✓Use :param_name style in SQL. better-sqlite3 binds them automatically.
14

procedure() — same pattern as table()

Access reports the same way you access tables. Returns a cached ProcedureHandle — same instance every call.

usagejs
const sewa = new Sutram({ db, schema,
  reports: require('./reports'),
});

// run with custom params
const rows = sewa.procedure('monthly_sales').run({
  from: '2024-06-01',
  to:   '2024-06-30',
});

// run with defaults
const rows = sewa.procedure('low_stock').run();

// inspect
sewa.procedure('monthly_sales').getParams();
sewa.procedure('monthly_sales').getMeta();

// cached — same instance
sewa.procedure('low_stock') === sewa.procedure('low_stock')  // true

// await safe — sync under hood
const rows = await sewa.procedure('low_stock').run();  // works fine
MethodReturns
.run(params?)Array of rows — sync
.getParams()Params definition array
.getMeta(){ name, label, report_type, params }
◈sutramCore only supports report_type: 'cte'. If you pass a report with report_type: 'procedure', it logs a warning and skips it. Use sutramcore-mysql for stored procedures.

Transactions

automatic + manualjs
// automatic — commit on success, rollback on throw
const id = sewa.transaction(() => {
  const oid = order.insert({ customer_id: 5 }, false);
  for (const item of cart) {
    orderItem.insert({ order_id: oid, ...item }, false);
  }
  return oid;
});

// manual
sewa.begin();
try {
  order.insert(data, false);
  sewa.commit();
} catch(e) {
  sewa.rollback();
  throw e;
}
